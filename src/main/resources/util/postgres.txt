CREATE TABLE IF NOT EXISTS post (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY,
  name varchar(64) NOT NULL,
  image_url varchar(512) NOT NULL,
  -- "DESCRIPTION" varchar(192) NOT NULL,
  description text[] NOT NULL,
  CONSTRAINT pk_post_id PRIMARY KEY (id),
  CONSTRAINT uk_post_name UNIQUE (name)
);

CREATE TABLE IF NOT EXISTS tag (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY,
  name varchar(64) NOT NULL,
  post_id INTEGER NOT NULL,
  CONSTRAINT pk_tag_id PRIMARY KEY (id),
  CONSTRAINT uk_tag_name_post_id UNIQUE (name, post_id),
  CONSTRAINT fk_tag_post_id FOREIGN KEY (post_id)
          REFERENCES post (id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS likes (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY,
  count INTEGER NOT NULL,
  post_id INTEGER NOT NULL,
  CONSTRAINT pk_likes_id PRIMARY KEY (id),
  CONSTRAINT fk_likes_post_id FOREIGN KEY (post_id)
          REFERENCES post (id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS comment (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY,
  post_comment varchar(1024),
  post_id INTEGER NOT NULL,
  CONSTRAINT pk_comment_id PRIMARY KEY (id),
  CONSTRAINT fk_comment_post_id FOREIGN KEY (post_id)
          REFERENCES post (id) ON DELETE CASCADE
);

SELECT post.id, post.name, image_url, SUBSTRING(description[1], 1, 7) AS abbreviatedDescription, ARRAY(SELECT name FROM tag
                                                                                                      WHERE post.id = tag.post_id
                                                                                                      ) AS tags
                                                                                                      , likes.count AS likesCount,
                                                                                                      COUNT(comment.post_id) AS commentCount
FROM post
LEFT JOIN likes ON post.id = likes.post_id
--LEFT JOIN likes USING (id)--
LEFT JOIN comment ON post.id = comment.post_id
GROUP BY comment.post_id, post.name, post.id, likes.count
