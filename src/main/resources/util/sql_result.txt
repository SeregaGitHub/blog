CREATE TABLE IF NOT EXISTS post (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY,
  name varchar(64) NOT NULL,
  image_url varchar(512) NOT NULL,
  -- "DESCRIPTION" varchar(192) NOT NULL,
  description text[] NOT NULL,
  CONSTRAINT pk_post_id PRIMARY KEY (id)--,
  --CONSTRAINT uk_post_name UNIQUE (name)  -- ???
);

CREATE TABLE IF NOT EXISTS tag (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY,
  name varchar(64) NOT NULL,
  --post_id INTEGER NOT NULL,
  CONSTRAINT pk_tag_id PRIMARY KEY (id),
  --CONSTRAINT uk_tag_name_post_id UNIQUE (name, post_id),
  CONSTRAINT uk_tag_name UNIQUE (name)--,
  -- CONSTRAINT fk_tag_post_id FOREIGN KEY (post_id)
  --         REFERENCES post (id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS post_tag (
	post_id INTEGER,
	tag_id INTEGER,
	CONSTRAINT pk_post_tag_id PRIMARY KEY (post_id, tag_id),
	CONSTRAINT fk_post_tag_post_id FOREIGN KEY (post_id)
          REFERENCES post (id) ON DELETE CASCADE,
	CONSTRAINT fk_post_tag_tag_id FOREIGN KEY (tag_id)
          REFERENCES tag (id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS likes (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY,
  count INTEGER NOT NULL,
  post_id INTEGER NOT NULL,
  CONSTRAINT pk_likes_id PRIMARY KEY (id),
  CONSTRAINT uk_likes_post_id UNIQUE (post_id),
  CONSTRAINT fk_likes_post_id FOREIGN KEY (post_id)
          REFERENCES post (id) ON DELETE CASCADE
); -- create index post_id, count ???

CREATE TABLE IF NOT EXISTS comment (
  id INTEGER GENERATED BY DEFAULT AS IDENTITY,
  post_comment varchar(2048),
  post_id INTEGER NOT NULL,
  CONSTRAINT pk_comment_id PRIMARY KEY (id),
  CONSTRAINT fk_comment_post_id FOREIGN KEY (post_id)
          REFERENCES post (id) ON DELETE CASCADE
); -- create index post_id, post_comment ???

-- Post Feed
SELECT p.id, p.name, p.image_url, SUBSTRING(p.description[1], 1, 7) AS abbreviatedDescription,
COUNT(c.post_id) AS commentCount, COALESCE (l.count, 0) AS likesCount, COALESCE (STRING_AGG(DISTINCT t.name, ' '), '') AS tags
FROM post p
LEFT JOIN comment c ON c.post_id = p.id
LEFT JOIN likes l ON l.post_id = p.id
LEFT JOIN post_tag pt ON p.id = pt.post_id
LEFT JOIN tag t ON t.id = pt.tag_id
GROUP BY p.id, p.name, c.post_id, l.count
ORDER BY p.id DESC
OFFSET 0
LIMIT 20;

-- create new post
CREATE PROCEDURE create_new_post(IN post_name varchar, IN image_url varchar, IN description text[], IN tags_array varchar[])
LANGUAGE plpgsql
AS $$
DECLARE
  p_id INT;
  tags_id_arr int[];
BEGIN
  INSERT INTO post(name, image_url, description)
	VALUES
	(post_name, image_url, description)
	--('nnn', 'uuu', ARRAY['description-51', 'description-52', 'description-53'])
	ON CONFLICT DO NOTHING
	RETURNING id INTO p_id;

	INSERT INTO tag(name)
	VALUES (unnest(tags_array))
	ON CONFLICT DO NOTHING;

	SELECT ARRAY_AGG(t.id) FROM tag t WHERE name LIKE ANY(tags_array) INTO tags_id_arr;

	INSERT INTO post_tag
	VALUES
	-- (p_id, unnest(ARRAY[3, 4, 19, 20]));
	(p_id, unnest(tags_id_arr));
END;
$$;

-- filtering by tag
SELECT p.id, p.name, p.image_url, SUBSTRING(p.description[1], 1, 7) AS abbreviatedDescription,
COUNT(c.post_id) AS commentCount, COALESCE (l.count, 0) AS likesCount, COALESCE (STRING_AGG(DISTINCT t.name, ' '), '') AS tags
FROM post p
LEFT JOIN comment c ON c.post_id = p.id
LEFT JOIN likes l ON l.post_id = p.id
LEFT JOIN post_tag pt ON p.id = pt.post_id
LEFT JOIN tag t ON t.id = pt.tag_id
WHERE t.name = '#rise'
GROUP BY p.id, p.name, c.post_id, l.count
ORDER BY p.id DESC;

------------------------------------------------------------------------------------------------------------------------

-- select one post
SELECT p.id, p.name, p.image_url, p.description,
		COALESCE (STRING_AGG(DISTINCT t.name, ' '), '') AS tags,
		COALESCE (l.count, 0) AS likesCount,
		CASE
    		WHEN length((ARRAY_AGG(DISTINCT c.post_comment))[1]) > 0 THEN (ARRAY_AGG(DISTINCT c.post_comment))
    		ELSE ARRAY['']
		END AS comments
FROM post p
LEFT JOIN comment c ON c.post_id = p.id
LEFT JOIN likes l ON l.post_id = p.id
LEFT JOIN post_tag pt ON p.id = pt.post_id
LEFT JOIN tag t ON t.id = pt.tag_id
WHERE p.id = 1
GROUP BY p.id, p.name, l.count;
